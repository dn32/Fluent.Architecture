   public enum EnumCacheScope
    {
        REQUEST,
        USER,
        TENANT,
        SYSTEM
    }

    public class Cache
    {
        public string Id { get; set; }

        public EnumCacheScope CacheScope { get; set; }

        public Cache(string id, EnumCacheScope cacheScope)
        {
            Id = id;
            CacheScope = cacheScope;
        }

        public virtual object Get()
        {
            return default;
        }
    }

    public class Cache<T> : Cache
    {
        public Cache(string id, EnumCacheScope cacheScope) : base(id, cacheScope)
        {
        }

        public new T Get()
        {
            return (T)base.Get();
        }
    }

    public static class CacheAgent
